# Network Reconnaissance Simulation
# Tool: Nmap
# Attacker: Kali Linux
# Target: Windows Endpoint


# Basic TCP SYN Scan

nmap -sS 192.168.81.138


# Scan Specific Ports

nmap -sS -p 22,80,135,445,3389 192.168.81.138


# Service Version Detection

nmap -sV 192.168.81.138


# Aggressive Scan (Lab Testing Only)

nmap -A 192.168.81.138


# Port Range Scan

nmap -p 1-1000 192.168.81.138


# Save Scan Results

nmap -sS -p 1-1000 -oN nmap_scan_result.txt 192.168.81.138
